% ---------- 代码 ----------
% minted 用 Pygments 高亮，通过 latexminted 调用（见 .latexmkrc 和 tools/latexminted）
\usepackage[newfloat]{minted}
\setminted{
  fontsize=\small,
  linenos,
  numbersep=6pt,
  frame=single,
  rulecolor=gray!50,
  bgcolor=gray!5,
  style=default,
  tabsize=4,
  breaklines,
}
\SetupFloatingEnvironment{listing}{name=代码, listname=代码清单, within=chapter}

% 带题注的代码文件：\codefile[语言]{文件}{题注}{标签}
% 不使用浮动体，长代码可以跨页；语言默认为 c
% \needspace：本页剩余空间放不下题注和几行代码时，先换页，免得题注单独留在页底
\usepackage{needspace}
\newcommand{\codefile}[4][c]{%
  \par\addvspace{\medskipamount}%
  \needspace{6\baselineskip}%
  \begingroup
    \captionsetup{type=listing}%
    \caption{#3}\label{#4}%
  \endgroup
  \nopagebreak[4]%
  \inputminted{#1}{#2}%
  \par\addvspace{\medskipamount}%
}
